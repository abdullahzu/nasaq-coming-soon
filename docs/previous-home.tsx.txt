export default function Home() {
  const [language, setLanguage] = useState<Language>("ar")
  const [projectType, setProjectType] = useState("")
  const [formStatus, setFormStatus] = useState("")
  const [formStatusType, setFormStatusType] = useState<"idle" | "sending" | "success" | "error">("idle")
  const [isSubmitting, setIsSubmitting] = useState(false)
  const projectTypeRef = useRef<HTMLButtonElement>(null)
  const t = content[language]
  const isArabic = language === "ar"
  const DirectionArrow = isArabic ? ArrowUpLeft : ArrowUpRight

  useEffect(() => {
    document.documentElement.lang = language
    document.documentElement.dir = isArabic ? "rtl" : "ltr"
  }, [isArabic, language])

  function toggleLanguage() {
    setLanguage((current) => (current === "ar" ? "en" : "ar"))
    setFormStatus("")
    setFormStatusType("idle")
  }

  async function submitProject(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()

    if (!projectType) {
      setFormStatus(t.contact.missingType)
      setFormStatusType("error")
      projectTypeRef.current?.focus()
      return
    }

    const form = event.currentTarget
    const data = new FormData(form)
    const selectedType = projectOptions[Number(projectType)]?.[language] ?? ""

    setIsSubmitting(true)
    setFormStatus(t.contact.sending)
    setFormStatusType("sending")

    try {
      const response = await fetch("/api/contact", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          name: String(data.get("name") || ""),
          company: String(data.get("company") || ""),
          projectType: selectedType,
          location: String(data.get("location") || ""),
          contact: String(data.get("contact") || ""),
          note: String(data.get("note") || ""),
          language,
          website: String(data.get("website") || ""),
        }),
      })

      if (!response.ok) throw new Error("Contact request failed")

      setFormStatus(t.contact.success)
      setFormStatusType("success")
      form.reset()
      setProjectType("")
    } catch {
      setFormStatus(t.contact.error)
      setFormStatusType("error")
    } finally {
      setIsSubmitting(false)
    }
  }

  return (
    <div className="site-shell" dir={isArabic ? "rtl" : "ltr"}>
      <a className="skip-link" href="#main-content">{isArabic ? "انتقل إلى المحتوى" : "Skip to content"}</a>

      <header className="site-header">
        <div className="container header-inner">
          <a className="brand" href="#top" aria-label={isArabic ? "نَسَق، الصفحة الرئيسية" : "NASAQ, home"}>
            <span className="brand-ar" lang="ar">نَسَق</span>
            <span className="brand-divider" aria-hidden="true" />
            <span className="brand-en" dir="ltr"><strong>NASAQ</strong><small>Foodservice Design &amp; Advisory</small></span>
          </a>

          <nav className="desktop-nav" aria-label={isArabic ? "التنقل الرئيسي" : "Primary navigation"}>
            <a href="#approach">{t.nav.approach}</a>
            <a href="#services">{t.nav.services}</a>
            <a href="#private">{t.nav.private}</a>
            <a href="#contact">{t.nav.contact}</a>
          </nav>

          <div className="header-actions">
            <Button type="button" variant="ghost" className="language-switch" onClick={toggleLanguage} aria-label={t.switchLabel}><span aria-hidden="true">{isArabic ? "EN" : "عربي"}</span></Button>
            <Button asChild className="header-cta"><a href="#contact">{t.discuss}</a></Button>
          </div>
        </div>
      </header>

      <main id="main-content">
        <section className="hero-section" id="top" aria-labelledby="hero-heading">
          <div className="hero-grid-lines" aria-hidden="true" />
          <div className="container hero-grid">
            <div className="hero-copy">
              <div className="status-line"><span aria-hidden="true" /><span>{t.status}</span></div>
              <p className="kicker">{t.hero.kicker}</p>
              <h1 id="hero-heading"><span>{t.hero.titleA}</span><span>{t.hero.titleB}</span></h1>
              <p className="hero-brand-line" dir={isArabic ? "rtl" : "ltr"}>{t.hero.slogan}</p>
              <p className="hero-body">{t.hero.body}</p>
              <div className="hero-actions">
                <Button asChild size="lg" className="primary-button"><a href="#contact">{t.hero.primary}<DirectionArrow aria-hidden="true" /></a></Button>
                <Button asChild size="lg" variant="outline" className="secondary-button"><a href="#approach">{t.hero.secondary}<ArrowDown aria-hidden="true" /></a></Button>
              </div>
              <p className="hero-audience">{t.hero.audience}</p>
            </div>

            <figure className="hero-visual" aria-labelledby="hero-visual-caption">
              <div className="hero-image-wrap">
                {/* eslint-disable-next-line @next/next/no-img-element */}
                <img src="/nasaq-full-counter-hero-v2.webp" width="1536" height="1024" fetchPriority="high" decoding="async" alt={t.hero.visualAlt} />
                <span className="image-rule" aria-hidden="true" />
              </div>
              <figcaption id="hero-visual-caption"><strong>{t.hero.visualTitle}</strong><span>{t.hero.visualNote}</span></figcaption>
            </figure>
          </div>
        </section>

        <section className="flow-section" aria-label={isArabic ? "رحلة القرار" : "Decision journey"}>
          <div className="container flow-inner"><p>{t.flow.label}</p><ul>{t.flow.items.map((item) => <li key={item}>{item}</li>)}</ul></div>
        </section>

        <section className="section approach-section" id="approach" aria-labelledby="approach-heading">
          <div className="container approach-grid">
            <div className="section-intro"><p className="kicker">{t.approach.kicker}</p><h2 id="approach-heading">{t.approach.title}</h2><p>{t.approach.body}</p></div>
            <ul className="principles-grid">{t.approach.points.map((point) => <li key={point.title}><span className="principle-mark" aria-hidden="true" /><h3>{point.title}</h3><p>{point.body}</p></li>)}</ul>
          </div>
        </section>

        <section className="section services-section" id="services" aria-labelledby="services-heading">
          <div className="services-grid-lines" aria-hidden="true" />
          <div className="container services-grid">
            <div className="services-intro"><p className="kicker kicker-light">{t.services.kicker}</p><h2 id="services-heading">{t.services.title}</h2><p>{t.services.body}</p><p className="services-note">{t.services.note}</p></div>
            <Accordion className="services-accordion" type="single" collapsible defaultValue="service-0">
              {t.services.items.map((service, index) => (
                <AccordionItem value={`service-${index}`} key={service.title}>
                  <AccordionTrigger className="service-trigger"><span className="service-name"><strong>{service.title}</strong><small dir={isArabic ? "ltr" : "rtl"}>{service.secondary}</small></span></AccordionTrigger>
                  <AccordionContent className="service-content"><p>{service.body}</p><div className="service-result"><span aria-hidden="true"><Check /></span><span><small>{t.services.resultLabel}</small><strong>{service.result}</strong></span></div></AccordionContent>
                </AccordionItem>
              ))}
            </Accordion>
          </div>
        </section>

        <section className="audience-section" aria-labelledby="audience-heading">
          <div className="container audience-inner"><p className="kicker">{t.audience.kicker}</p><h2 id="audience-heading">{t.audience.title}</h2><p>{t.audience.body}</p></div>
        </section>

        <section className="section private-section" id="private" aria-labelledby="private-heading">
          <div className="container private-grid">
            <figure className="private-visual">
              <div className="private-image-wrap">
                {/* eslint-disable-next-line @next/next/no-img-element */}
                <img src="/nasaq-villa-service-trolley-layout.webp" width="1536" height="1024" loading="lazy" decoding="async" alt={t.private.imageAlt} />
              </div>
              <figcaption>{t.private.caption}</figcaption>
            </figure>
            <div className="private-copy"><p className="kicker">{t.private.kicker}</p><h2 id="private-heading">{t.private.title}</h2><p className="private-body">{t.private.body}</p><ul className="private-highlights">{t.private.highlights.map((item) => <li key={item}>{item}</li>)}</ul><p className="private-note">{t.private.note}</p><Button asChild variant="outline" size="lg" className="private-cta"><a href="#contact">{t.hero.primary}<DirectionArrow aria-hidden="true" /></a></Button></div>
          </div>
        </section>

        <section className="contact-section" id="contact" aria-labelledby="contact-heading">
          <div className="container contact-grid">
            <div className="contact-intro"><p className="kicker kicker-light">{t.contact.kicker}</p><h2 id="contact-heading">{t.contact.title}</h2><p>{t.contact.body}</p><div className="delivery-destination"><span className="email-icon" aria-hidden="true"><Mail /></span><span><small>{t.contact.direct}</small><strong dir="ltr">{siteConfig.contactEmail}</strong><em>{t.contact.team}</em></span></div></div>

            <form className="project-form" onSubmit={submitProject}>
              <Input className="contact-honeypot" id="website" name="website" tabIndex={-1} autoComplete="off" aria-hidden="true" />
              <div className="form-row"><div className="field-group"><Label htmlFor="name">{t.contact.fields.name}</Label><Input id="name" name="name" placeholder={t.contact.placeholders.name} autoComplete="name" required /></div><div className="field-group"><Label htmlFor="company">{t.contact.fields.company}</Label><Input id="company" name="company" placeholder={t.contact.placeholders.company} autoComplete="organization" required /></div></div>
              <div className="form-row">
